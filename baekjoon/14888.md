### 문제 분석 요약
재귀를 이용하여 모든 경우의 수를 확인하는 문제

### 알고리즘 설계
조건
- 2 <= N <= 11, 1 <= Ai <= 100, |식의 중간 결과| <= 1,000,000,000
- 음수 나누기 양수의 몫을 구하는 경우 양수로 바꾸어 몫을 취한 뒤 음수로 바꾼다.

1. **함수 정의**
   ```js
   const numbers   // 수열
   const operators // 연산자 배열 ex. [ '+', '+', '+', '-', '/' ]
   const isUsed    // 위 연산자 배열의 연산자 사용 여부 배열
   let max = -1000000000;
   let min = 1000000000;
   
   function recursive(index, total) { /* ... */ }
   ```
2. **Base Condition**

   현재 원소 인덱스가 N과 같다면 마지막 원소까지 탐색을 완료한 경우

   이때 현재까지의 계산 결과가
   - max보다 크다면 최댓값 업데이트
   - min보다 작다면 최솟값 업데이트
4. **재귀 식**
   ```js
   function recursive(index, total) {
       Base Condition

       남은 연산자들에 대해
       연산자가 "+" 라면
       recursive(index + 1, total + numbers[index]);
       연산자가 "-" 라면
       recursive(index + 1, total - numbers[index]);
       연산자가 "*" 라면
       recursive(index + 1, total * numbers[index]);
       연산자가 "/" 라면
       recursive(
         index + 1,
         total < 0 && numbers[index] > 0
           ? -Math.floor(-total / numbers[index])
           : Math.floor(total / numbers[index])
       );
   }
   ```

### 코드
```js
const fs = require("fs");
const rawInput = fs.readFileSync("input").toString().trim(); // fs.readFileSync(0, "utf-8").trim();

function solution() {
  const [, numbers, operatorCounts] = rawInput
    .split("\n")
    .map((row) => row.split(" ").map(Number));
  const operators = [];
  for (let i = 0; i < operatorCounts.length; i++) {
    for (let j = 0; j < operatorCounts[i]; j++) operators.push(i);
  }
  const isUsed = operators.map(() => false);

  let max = -1000000000;
  let min = 1000000000;

  function recursive(index, total) {
    if (index === numbers.length) {
      if (total > max) max = total;
      if (total < min) min = total;
      return;
    }

    for (let i = 0; i < operators.length; i++) {
      if (!isUsed[i]) {
        isUsed[i] = true;
        switch (operators[i]) {
          case 0:
            recursive(index + 1, total + numbers[index]);
            break;
          case 1:
            recursive(index + 1, total - numbers[index]);
            break;
          case 2:
            recursive(index + 1, total * numbers[index]);
            break;
          case 3:
            recursive(
              index + 1,
              total < 0 && numbers[index] > 0
                ? -Math.floor(-total / numbers[index])
                : Math.floor(total / numbers[index])
            );
            break;
        }
        isUsed[i] = false;
      }
    }
  }

  recursive(1, numbers[0]);
  console.log(`${max}\n${min}`);
}

solution();
```

### 시간 복잡도
O(N!), 재귀함수가 호출 되는 횟수가 1씩 줄어들기 때문

### 느낀점 or 기억할 정보
최댓값을 구할 때 가장 큰 수부터 곱셈을 할당하는 방식으로 풀어보려고 했는데, 중간 결과가 음수가 나올 수 있어서 그렇게 풀 수 없음.
